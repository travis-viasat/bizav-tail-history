**Step D — Create tail-history/src/pages/tailHistory/charts/CustomChart/useCustomChartMock.ts:**
Full Viasat copyright header.

```typescript
import {useMemo} from 'react';
import type Highcharts from 'highcharts/highstock';
import {generateMockSeries} from '../../__mocks__/chartData';
import {CUSTOM_CHART_COLORS} from '../../../../theme/colors';
import type {ChartType} from '../../../../catalog/metricCatalog';

export interface CustomChartMockResult {
  series: Highcharts.SeriesOptionsType[];
  isLoading: boolean;
  isEmpty: boolean;
  error: Error | null;
}

/**
 * Generic mock hook for custom chart data.
 * Derives a numeric seed from metricKey so each metric gets visually distinct data.
 * Color is assigned deterministically via seed % CUSTOM_CHART_COLORS.length.
 * Returns the same shape as chart-specific mock hooks (isLoading, isEmpty, error).
 */
export function useCustomChartMock(
  metricKey: string,
  chartType: ChartType
): CustomChartMockResult {
  const seed = metricKey.split('').reduce((acc, ch) => acc + ch.charCodeAt(0), 0);
  const color = CUSTOM_CHART_COLORS[seed % CUSTOM_CHART_COLORS.length];

  const series: Highcharts.SeriesOptionsType[] = useMemo(
    () => [
      {
        type: chartType as Highcharts.SeriesOptionsType['type'],
        name: metricKey,
        data: generateMockSeries(seed, 0, 100),
        color,
      },
    ],
    [metricKey, chartType, seed, color]
  );

  return {series, isLoading: false, isEmpty: false, error: null};
}
```
